\section{Probability tricks (conditioning!)}
	\begin{align*} 
  		P(A|C)=\sum_kP(A|B_k,C)\,P(B_k\,|\,C)
  \end{align*}
 \textbf{weights carry $C$ too}. $P(B_k|C)=P(B_k)$ only if $B_k\perp C$
  \begin{align*} 
	  P(B|A)=\tfrac{P(A|B)P(B)}{P(A)}\\
	  P(A)=\sum_kP(A|B_k)P(B_k)
  \end{align*}
\begin{itemize}
  \item $P(A\cap B|C)=P(A|B,C)P(B|C)$
  \item \textbf{Safe order}: condition on the \emph{source} first (Alice's state/bit/basis), then on later random events (Eve's outcome, Bob's basis). Bases are independent of bits
  \item Ex.\ Breidbart: $P(x=0\,|\,\text{Eve sees }\tfrac\pi8)=\cos^2\tfrac\pi8\neq\tfrac12$
  \item $\pm1$ variables: $E[xy]=P(x{=}y)-P(x{\ne}y)=2P(x{=}y)-1$
  \item Error rate when fraction $p$ attacked with error $q$: $pq$
  \item Detect with $n$ test bits, per-bit error $q$: $1-(1-q)^n\ge1-\delta\iff n\ge\tfrac{\ln\delta}{\ln(1-q)}$ ($q=\tfrac14,\delta=10^{-6}$: $n=49$)
  \item Uniform choice among $k$ options: $\tfrac1k$ each; prob.\ bases match $=\tfrac12$ (BB84), $\tfrac29$ equal angles (Ekert)
\end{itemize}

\section{Dirac / linear algebra tricks}
\begin{itemize}
  \item $\braket{\psi}{\phi}=\sum_i\psi_i^*\phi_i$: \textbf{linear in the ket}, conj.-linear in the bra. $(a\ket{x}+b\ket{y})^\dagger=a^*\bra{x}+b^*\bra{y}$
  \item $\braket{\psi}{\phi}^*=\braket{\phi}{\psi}$, \ $\mel{\psi}{A}{\phi}^*=\mel{\phi}{A^\dagger}{\psi}$, \ $(AB)^\dagger=B^\dagger A^\dagger$
  \item $A=\sum_{ij}A_{ij}\dyad{i}{j}$, $A_{ij}=\mel{i}{A}{j}$; \ $\dyad{a}{b}\,\ket{c}=\braket{b}{c}\ket{a}$
  \item Completeness: $\sum_i\dyad{v_i}=I$ for any ONB $\Rightarrow$ $\ket{\psi}=\sum_i\braket{v_i}{\psi}\ket{v_i}$ (change of basis)
  \item Spectral: $A=\sum\lambda_i\dyad{v_i}$, $f(A)=\sum f(\lambda_i)\dyad{v_i}$. Hermitian: $\lambda\in\mathbb R$, ONB of eigvecs. Unitary: $\abs{\lambda}=1$, preserves $\braket{\cdot}{\cdot}$
  \item Build $U$ from its action: $U=\sum_i \dyad{\text{out}_i}{\text{in}_i}$ (in$_i$ ONB)
  \item $(\hat n\cdot\vec\sigma)^2=I\Rightarrow e^{i\theta\,\hat n\cdot\vec\sigma}=\cos\theta\,I+i\sin\theta\,\hat n\cdot\vec\sigma$
  \item Projector $P=\dyad{v}$: $P^2=P=P^\dagger$, $\norm{P\ket\psi}^2=\abs{\braket{v}{\psi}}^2$
\end{itemize}

\section{BB84}
\begin{itemize}
  \item Alice: random bits $x_i$, bases $e_i$; sends $H^{e_i}\ket{x_i}\in\{\ket0,\ket1,\pp,\mm\}$
  \item Bob: random $d_i$, measures in $\{H^{d_i}\ket0,H^{d_i}\ket1\}$, gets $y_i$
  \item Public: announce $e_i,d_i$ (not $x,y$). \textbf{Sift}: keep $e_i=d_i$ ($\approx N/2$)
  \item Check: reveal $\approx\varepsilon N/2$ sifted bits, compare, discard them. Too many errors $\Rightarrow$ abort
  \item Lemma: $P(x{=}y|e{=}d)=1$; $P(x{=}y|e{\ne}d)=\tfrac12$
  \item \textbf{Intercept-resend} (Eve random basis $E_i$, resends $H^{E_i}\ket{y^{Eve}}$): $P_{Eve}(x\ne y|e{=}d)=\tfrac12\cdot0+\tfrac12\cdot\tfrac12=\tfrac14$. Eve knows bit w.p.\ $\tfrac34$ (certain if $E_i=e_i$, info after basis announcement)
  \item Fraction $p$ attacked: QBER $\tfrac p4$
  \item \textbf{Breidbart}: Eve measures $\{\ket{\tfrac\pi8},\ket{\tfrac{5\pi}8}\}$, each of Alice's states at $\tfrac\pi8$ from one vector: QBER $2\cos^2\tfrac\pi8\sin^2\tfrac\pi8=\tfrac14$, Eve correct $\cos^2\tfrac\pi8\approx0.854$
  \item Copy-and-resend impossible (no-cloning, states non-orthogonal)
  \item Noise must be low enough to be distinguished from Eve
\end{itemize}

\section{B92 (N\&C 12.6 version)}
Alice generates a random sequence $e_1, \ldots, e_N$ of bits kept secret. She sends $\ket{0}$ if $e_i = 0$ and $H\ket{0} = \ket{+}$ if $e_i = 1$, she sends $H^{e_i}\ket{0}$

Bob generates a random sequence $d_1, \ldots d_N$, he measured the qbit received with the basis $\left\{\ket{0}, \ket{1}\right\}$ ($Z$ basis) if $d_i = 0$ and $\left\{H\ket{0}, H\ket{1}\right\}$ if $d_i = 1$.

Bob announces on a public channel his measurement outcome $y_1, \ldots, y_N$. One can show that $P\left(y_i = 0| e_i = d_i\right) = 1, P\left(y_i = 0| e_i \neq d_1\right) = \frac{1}{2}$. We can deduce from the fours probabilities that $P\left(e_i = 1 - d_i| y_i = 1\right) = 1$, Hence when $y_i = 1$ Bob switches his bit.

\section{CHSH / Bell inequalities}

	\ket{B_{00}} = \frac{1}{\sqrt{2}}\left(\ket{\alpha \alpha} + \ket{\alpha_\perp \alpha_\perp} \right)

  \item $\B{11}$: $E=-\cos2(\alpha-\beta)$

  \item Noise: w.p.\ $p$ Bell pair, else random bits: $E\to pE$, violation iff $p>\tfrac1{\sqrt2}$

\section{Ekert 91}
\begin{itemize}
  \item Each round a $\B{00}$ pair. Alice $\alpha\in\{-\tfrac\pi4,-\tfrac\pi8,0\}$, Bob $\beta\in\{-\tfrac\pi8,0,\tfrac\pi8\}$, uniform; record $\pm1$
  \item Announce settings (not outcomes). 9 combos, each $\tfrac19$
  \item \textbf{Test} (4/9): $(\alpha_3,\beta_3),(\alpha_3,\beta_1),(\alpha_1,\beta_1),(\alpha_1,\beta_3)$; reveal outcomes, compute $S=E_{33}+E_{31}+E_{11}-E_{13}=4\cdot\tfrac1{\sqrt2}=2\sqrt2$

  \item Eve measures both in $Z$ \& resends $\ket{00}/\ket{11}$: $E=\cos2\alpha\cos2\beta$, $S=\sqrt2<2$ $\Rightarrow$ detected
  \item Then error correction + privacy amplification (shorter key the further $S$ is below $2\sqrt2$)
\end{itemize}
